**Goal:** Build a Go client library at `go/client/` that lets WorkFort services discover each other via Pylon's `GET /api/services` endpoint.

**Architecture:** Stdlib-only HTTP client following the Hive client pattern (`hive/client/`). A `Client` struct wraps `net/http` with Bearer auth, provides `Services()` and `ServiceByName()` methods. No caching in v1.

**Tech Stack:** Go standard library only (`net/http`, `encoding/json`, `errors`). No external dependencies.

```go
package client

import (
	"errors"
	"fmt"
)

var (
	// ErrNotFound is returned by ServiceByName when the named service
	// is not in the registry.
	ErrNotFound = errors.New("service not found")

	// ErrUnauthorized is returned when Pylon responds with 401.
	ErrUnauthorized = errors.New("unauthorized")
)

// APIError is returned for non-2xx HTTP responses from Pylon.
type APIError struct {
	StatusCode int
	Message    string
	sentinel   error
}

func (e *APIError) Error() string {
	if e.Message != "" {
		return fmt.Sprintf("pylon api error %d: %s", e.StatusCode, e.Message)
	}
	return fmt.Sprintf("pylon api error %d", e.StatusCode)
}

// Unwrap allows errors.Is to match sentinel errors.
func (e *APIError) Unwrap() error { return e.sentinel }
```

```go
// SPDX-License-Identifier: GPL-3.0-or-later
package client

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"strings"
	"time"
)

// Client is an HTTP client for the Pylon service registry API.
type Client struct {
	http    http.Client
	baseURL string
	token   string
}

// New creates a Pylon client.
// pylonURL is the base URL of the Pylon daemon (e.g., "http://pylon:18000").
// token is a Passport JWT or API key sent as a Bearer token.
func New(pylonURL, token string) *Client {
	return &Client{
		http:    http.Client{Timeout: 10 * time.Second},
		baseURL: strings.TrimRight(pylonURL, "/"),
		token:   token,
	}
}

// Services returns all registered services from Pylon.
func (c *Client) Services(ctx context.Context) ([]Service, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.baseURL+"/api/services", nil)
	if err != nil {
		return nil, fmt.Errorf("build request: %w", err)
	}
	req.Header.Set("Accept", "application/json")
	if c.token != "" {
		req.Header.Set("Authorization", "Bearer "+c.token)
	}

	resp, err := c.http.Do(req)
	if err != nil {
		return nil, fmt.Errorf("pylon request failed: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return nil, decodeAPIError(resp)
	}

	var out servicesResponse
	if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
		return nil, fmt.Errorf("decode response: %w", err)
	}
	return out.Services, nil
}

// decodeAPIError reads the response body and returns an *APIError.
func decodeAPIError(resp *http.Response) *APIError {
	var body struct {
		Error string `json:"error"`
	}
	json.NewDecoder(resp.Body).Decode(&body) //nolint:errcheck

	ae := &APIError{StatusCode: resp.StatusCode, Message: body.Error}
	if resp.StatusCode == http.StatusUnauthorized {
		ae.sentinel = ErrUnauthorized
	}
	return ae
}
```

```go
// ServiceByName returns a specific service by its registered name.
// Returns ErrNotFound if the service is not in the registry.
func (c *Client) ServiceByName(ctx context.Context, name string) (*Service, error) {
	services, err := c.Services(ctx)
	if err != nil {
		return nil, err
	}
	for i := range services {
		if services[i].Name == name {
			return &services[i], nil
		}
	}
	return nil, fmt.Errorf("%w: %s", ErrNotFound, name)
}
```
